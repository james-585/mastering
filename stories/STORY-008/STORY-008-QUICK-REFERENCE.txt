    from suno_mastering import master
    from suno_mastering.config import MasteringConfig, StemConfig

    config = MasteringConfig(
        stem_config=StemConfig(
            enabled=True,
            model_name="htdemucs",
            bass_mono_cutoff_hz=90.0,
            vocal_lpf_hz=15000.0,
            vocal_hpf_hz=80.0,
        )
    )

    result = master("input.wav", config=config)
    print(result.stem_separation.separation_time_s)


┌──────────────────────────────────────────────────────────────────────────────┐
│ WHAT IT DOES                                                                 │
└──────────────────────────────────────────────────────────────────────────────┘

    Phase A: AI Separation (Demucs)
    ├─ Input: Stereo audio array
    └─ Output: 4 stems → vocals, drums, bass, other

    Phase B: Targeted DSP
    ├─ Bass:   Mono summing below 90 Hz (8th-order Butterworth lowpass)
    ├─ Vocals: Low-pass at 15 kHz + high-pass at 80 Hz
    └─ Drums/Other: Pass-through (no processing)

    Phase C: Re-Summation
    ├─ Sum: vocals + drums + bass + other
    ├─ Sanity checks: NaN/Inf detection, clipping warning
    └─ Output: Single stereo float64 array → passes to Stage 1 (measure_all)
